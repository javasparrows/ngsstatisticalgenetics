# コード7
# VCFファイルのQUAL列（6列目）が35以上のバリアントを抽出
java -jar SnpSift.jar filter "( QUAL >= 35 )" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.QUAL35.vcf
# VCFファイルのCHROM列（1列目）で染色体番号が22のバリアントを抽出
# この結果はSnpSift dbNSFPで使用する
java -jar SnpSift.jar filter "( CHROM = 'chr22' )" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.CHROM22.vcf
# VCFファイルのREF列（4列目）とALT列（5列目）で，
# transitionを起こしているバリアントを検出
## 核酸の変異で，ピリミジン（C,T/U）間，またはプリン（A,G）間の変異をtransition
## ピリミジンとプリン間での変異をtransversionとよぶ
## transversionはtransitionに比べ起こりにくい
java -jar SnpSift.jar filter "(( REF = 'A' ) & ( ALT = 'G' ))|(( REF = 'G' ) & ( ALT = 'A' ))|(( REF = 'C' ) & ( ALT = 'T' ))|(( REF = 'T' ) & ( ALT = 'C'))" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.Transition.vcf
# ANNフィールド内のAnnotation列（EFFECT）がmissense_variantのバリアントを抽出
java -jar SnpSift.jar filter "ANN[*].EFFECT has 'missense_variant'" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.missense.vcf
# ANNフィールド内のAnnotation Impact列（IMPACT）がHIGHのバリアントを抽出
java -jar SnpSift.jar filter "ANN[*].IMPACT has 'HIGH'" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.high_impact.vcf
# ANNフィールド内のGene Name列（GENE）がTRMT2Aのバリアントを抽出
java -jar SnpSift.jar filter "ANN[*].GENE has 'TRMT2A'" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.TRMT2A.vcf
# ANNフィールド内のTranscript_Biotype列（BIOTYPE）がprotein_codingのバリアントを抽出
java -jar SnpSift.jar filter "ANN[*].BIOTYPE has 'protein_coding'" ~/snpEff_latest_core/examples/merged.vqsr.ann.vcf > ~/snpEff_latest_core/examples/merged.vqsr.ann.filtered.protein_coding.vcf
