# コード0: Apple Silicon MacでSingularity/Apptainerのセットアップ
# Apple Silicon MacではSingularityをネイティブに実行できないため、Limaを使用してLinux環境を構築する

## Limaのインストール
brew install lima

## Lima VMの作成と起動（Ubuntu 22.04）
limactl start --name=singularity template://ubuntu
limactl create default

## VM内でApptainerをインストール
lima sudo apt-get update
lima sudo apt-get install -y software-properties-common
lima sudo add-apt-repository -y ppa:apptainer/ppa
lima sudo apt-get update
lima sudo apt-get install -y apptainer
# ↑2025/12/16 数秒かかった

## Apptainerの動作確認
lima apptainer --version

## 便利なエイリアスの設定（オプション）
# ~/.config/fish/config.fish に追加する場合（fishシェル）
echo "alias singularity='lima apptainer'" >> ~/.config/fish/config.fish
# ~/.bashrc または ~/.zshrc に追加する場合
# echo "alias singularity='lima apptainer'" >> ~/.bashrc

## 以降のsingularityコマンドは「lima apptainer」として実行する
# または、エイリアスを設定した後はそのまま「singularity」として実行可能

# コード3
# 1000人ゲノムプロジェクトの30X GRCh38版を4人分ダウンロード
mkdir -p cram
cd cram
wget -c ftp://ftp.sra.ebi.ac.uk/vol1/run/ERR324/ERR3240193/HG00118.final.cram
# ↑100GBくらいありそう、柏田のMacには厳しい

wget -c ftp://ftp.sra.ebi.ac.uk/vol1/run/ERR324/ERR3240146/HG00140.final.cram
wget -c ftp://ftp.sra.ebi.ac.uk/vol1/run/ERR324/ERR3240150/HG00145.final.cram
# 基準ゲノムのダウンロード
wget ftp://ftp.1000genomes.ebi.ac.uk/vol1/ftp/technical/reference/GRCh38_reference_genome/GRCh38_full_analysis_set_plus_decoy_hla.fa

# コード4
# 出力はmy-cohort-smoove.genotyped.vcf.gzとなる
# hg19/b37, hg38/b38のどちらかのproblematic regionをbed変数に格納する
# 今回はhg38なので，hg38の場合
reference_fasta=GRCh38_full_analysis_set_plus_decoy_hla.fa
bed=exclude.cnvnator_100bp.GRCh38.20170403.bed
# hg19の場合は以下のようにする
# reference_fasta=hs37d5.fa
# bed=ceph18.b37.lumpy.exclude.2014-01-15.bed
# CPUのコア数（ハイパースレッディング対応の場合はスレッド数）に応じて値を決める
threads=16

# コード5
# smooveの実行
smoove call --removepr \
--name my-cohort \
--exclude ${bed} \
--fasta ${reference_fasta} \
--processes ${threads} \
--genotype \
./cram/*.cram
